"""Violating shipped-script payload naming its Git vector through constants.

The same derivation as the literal payload, spelled the way the adapter spells
its own vector, so a detector that reads only string literals misses it.
"""

import subprocess

GIT_COMMAND = "git"
REV_PARSE_COMMAND = "rev-parse"
GIT_COMMON_DIR_OPTION = "--git-common-dir"

project_key = subprocess.run(
    [GIT_COMMAND, REV_PARSE_COMMAND, GIT_COMMON_DIR_OPTION],
    capture_output=True,
    text=True,
).stdout.strip()
